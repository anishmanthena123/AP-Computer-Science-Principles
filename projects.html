Shoe Shopping Adventure

My Python choices game with 20 questions. Pick between two options at each step of a trip to the mall and see how your shopping adventure ends!

The game uses Python functions, if/else statements, random numbers, and colorful text.

import time
import random


class fg:
    BLACK   = '\033[30m'
    RED     = '\033[31m'
    GREEN   = '\033[32m'
    YELLOW  = '\033[33m'
    BLUE    = '\033[34m'
    MAGENTA = '\033[35m'
    CYAN    = '\033[36m'
    WHITE   = '\033[37m'
    RESET   = '\033[39m'

class bg:
    BLACK   = '\033[40m'
    RED     = '\033[41m'
    GREEN   = '\033[42m'
    YELLOW  = '\033[43m'
    BLUE    = '\033[44m'
    MAGENTA = '\033[45m'
    CYAN    = '\033[46m'
    WHITE   = '\033[47m'
    RESET   = '\033[49m'

class style:
    BRIGHT    = '\033[1m'
    DIM       = '\033[2m'
    NORMAL    = '\033[22m'
    RESET_ALL = '\033[0m'


def masterFunction(question, option1, option2, resp1, resp2, color1, color2):
    answer = input(fg.BLACK + question + " " + color1 + option1 + fg.BLACK + " or " + color2 + option2 + fg.BLACK + "? " + fg.RESET)

    if answer.lower() == option1.lower():
        print(fg.GREEN + resp1 + fg.RESET)
        return option1

    elif answer.lower() == option2.lower():
        print(fg.BLUE + resp2 + fg.RESET)
        return option2

    else:
        print(fg.RED + "oops, invalid input" + fg.RESET)
        return masterFunction(question, option1, option2, resp1, resp2, color1, color2)
